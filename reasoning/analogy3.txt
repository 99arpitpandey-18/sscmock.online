function generateNumberAnalogy() {
    const a = Math.floor(Math.random() * 20) + 3;
    const b = a * (a + 1);
    let c = a + Math.floor(Math.random() * 10) + 2;
    if (c === a) c = a + 3;
    const d = c * (c + 1);
    const opts = shuffle([String(d), String(d + Math.floor(Math.random()*20)+5), String(d - Math.floor(Math.random()*20)-5), String(d + Math.floor(Math.random()*30)+10)]);
    const ansIdx = opts.indexOf(String(d));
    
    // Deep Explanation
    const exp_en = `🔍 Step-by-step reasoning:<br>
    <b>Step 1:</b> Identify the relationship in the first pair (${a} : ${b}).<br>
    <b>Step 2:</b> Here, the second number is obtained by multiplying the first number by its successor: ${a} × (${a}+1) = ${a} × ${a+1} = ${b}.<br>
    <b>Step 3:</b> Now apply the exact same logic to the second pair (${c} : ?).<br>
    <b>Step 4:</b> ${c} × (${c}+1) = ${c} × ${c+1} = ${d}.<br>
    ✅ <b>Therefore, the correct answer is ${d}.</b>`;
    
    const exp_hi = `🔍 चरणबद्ध हल (Step-by-step):<br>
    <b>चरण 1:</b> पहले युग्म (${a} : ${b}) में संबंध पहचानें।<br>
    <b>चरण 2:</b> यहाँ दूसरी संख्या, पहली संख्या को उसके अगले क्रम (successor) से गुणा करके प्राप्त होती है: ${a} × (${a}+1) = ${a} × ${a+1} = ${b}.<br>
    <b>चरण 3:</b> अब यही नियम दूसरे युग्म (${c} : ?) पर लागू करें।<br>
    <b>चरण 4:</b> ${c} × (${c}+1) = ${c} × ${c+1} = ${d}.<br>
    ✅ <b>अतः सही उत्तर ${d} है।</b>`;
    
    return { q_en: `${a} : ${b} :: ${c} : ?`, q_hi: `${a} : ${b} :: ${c} : ?`, opts_en: opts, opts_hi: opts, answer: ansIdx, type: "Number Analogy", exp_en, exp_hi };
}

function generateAlphabetAnalogy() {
    const alph = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
    const shift = Math.floor(Math.random() * 4) + 1;
    const start = Math.floor(Math.random() * 15) + 1;
    if (start + 3 >= 26) return generateAlphabetAnalogy();
    const p1 = alph[start] + alph[start+shift] + alph[start+2*shift];
    const p2 = alph[start+shift] + alph[start+2*shift] + alph[start+3*shift];
    const start2 = Math.floor(Math.random() * 12) + 8;
    if (start2 + 3 >= 26) return generateAlphabetAnalogy();
    const p3 = alph[start2] + alph[start2+shift] + alph[start2+2*shift];
    const p4 = alph[start2+shift] + alph[start2+2*shift] + alph[start2+3*shift];
    const opts = shuffle([p4, p3, p1, p2+'X']);
    const ansIdx = opts.indexOf(p4);
    
    const exp_en = `🔍 Step-by-step reasoning:<br>
    <b>Step 1:</b> Check the shift between corresponding letters of ${p1} and ${p2}.<br>
    <b>Step 2:</b> ${p1[0]} → ${p2[0]} (+${shift}), ${p1[1]} → ${p2[1]} (+${shift}), ${p1[2]} → ${p2[2]} (+${shift}).<br>
    <b>Step 3:</b> Apply the same +${shift} shift to ${p3}:<br>
    ${p3[0]} → ${String.fromCharCode(p3.charCodeAt(0)+shift)}, ${p3[1]} → ${String.fromCharCode(p3.charCodeAt(1)+shift)}, ${p3[2]} → ${String.fromCharCode(p3.charCodeAt(2)+shift)}.<br>
    ✅ <b>Therefore, the correct answer is ${p4}.</b>`;
    
    const exp_hi = `🔍 चरणबद्ध हल:<br>
    <b>चरण 1:</b> ${p1} और ${p2} के संगत अक्षरों के बीच अंतर (shift) देखें।<br>
    <b>चरण 2:</b> ${p1[0]} → ${p2[0]} (+${shift}), ${p1[1]} → ${p2[1]} (+${shift}), ${p1[2]} → ${p2[2]} (+${shift}).<br>
    <b>चरण 3:</b> यही +${shift} शिफ्ट ${p3} पर लागू करें:<br>
    ${p3[0]} → ${String.fromCharCode(p3.charCodeAt(0)+shift)}, ${p3[1]} → ${String.fromCharCode(p3.charCodeAt(1)+shift)}, ${p3[2]} → ${String.fromCharCode(p3.charCodeAt(2)+shift)}.<br>
    ✅ <b>अतः सही उत्तर ${p4} है।</b>`;
    
    return { q_en: `${p1} : ${p2} :: ${p3} : ?`, q_hi: `${p1} : ${p2} :: ${p3} : ?`, opts_en: opts, opts_hi: opts, answer: ansIdx, type: "Alphabet Analogy", exp_en, exp_hi };
}

function generateWordAnalogy() {
    const pair = wordPairs[Math.floor(Math.random() * wordPairs.length)];
    const opts_en = shuffle([pair.en[3], pair.en[0], pair.en[1], "None of these"]);
    const opts_hi = shuffle([pair.hi[3], pair.hi[0], pair.hi[1], "इनमें से कोई नहीं"]);
    const ansIdx = opts_en.indexOf(pair.en[3]);
    
    const exp_en = `🔍 Step-by-step reasoning:<br>
    <b>Step 1:</b> Identify the logical relationship between "${pair.en[0]}" and "${pair.en[1]}".<br>
    <b>Step 2:</b> "${pair.en[0]}" is associated with the action/place/thing "${pair.en[1]}". (Example: You read a book).<br>
    <b>Step 3:</b> Apply the exact same association to "${pair.en[2]}".<br>
    <b>Step 4:</b> The logical pair becomes "${pair.en[2]}" : "${pair.en[3]}".<br>
    ✅ <b>Therefore, the correct answer is "${pair.en[3]}".</b>`;
    
    const exp_hi = `🔍 चरणबद्ध हल:<br>
    <b>चरण 1:</b> "${pair.hi[0]}" और "${pair.hi[1]}" के बीच तार्किक संबंध पहचानें।<br>
    <b>चरण 2:</b> "${pair.hi[0]}" का संबंध क्रिया/स्थान/वस्तु "${pair.hi[1]}" से है। (जैसे: पुस्तक पढ़ी जाती है)।<br>
    <b>चरण 3:</b> यही संबंध "${pair.hi[2]}" पर लागू करें।<br>
    <b>चरण 4:</b> तार्किक युग्म बनेगा "${pair.hi[2]}" : "${pair.hi[3]}".<br>
    ✅ <b>अतः सही उत्तर "${pair.hi[3]}" है।</b>`;
    
    return { q_en: `${pair.en[0]} : ${pair.en[1]} :: ${pair.en[2]} : ?`, q_hi: `${pair.hi[0]} : ${pair.hi[1]} :: ${pair.hi[2]} : ?`, opts_en, opts_hi, answer: ansIdx, type: "Word Analogy", exp_en, exp_hi };
}

function generateMixedAnalogy() {
    const pair = mixedPairs[Math.floor(Math.random() * mixedPairs.length)];
    const opts_en = shuffle([pair.en[3], pair.en[0], pair.en[1], "None of these"]);
    const opts_hi = shuffle([pair.hi[3], pair.hi[0], pair.hi[1], "इनमें से कोई नहीं"]);
    const ansIdx = opts_en.indexOf(pair.en[3]);
    
    const exp_en = `🔍 Step-by-step reasoning:<br>
    <b>Step 1:</b> Analyze the relationship between "${pair.en[0]}" and "${pair.en[1]}".<br>
    <b>Step 2:</b> This is a general knowledge or functional relationship. (e.g., Pound is the currency of England).<br>
    <b>Step 3:</b> Apply the same functional relationship to "${pair.en[2]}".<br>
    <b>Step 4:</b> The correct counterpart for "${pair.en[2]}" is "${pair.en[3]}".<br>
    ✅ <b>Therefore, the correct answer is "${pair.en[3]}".</b>`;
    
    const exp_hi = `🔍 चरणबद्ध हल:<br>
    <b>चरण 1:</b> "${pair.hi[0]}" और "${pair.hi[1]}" के बीच संबंध का विश्लेषण करें।<br>
    <b>चरण 2:</b> यह सामान्य ज्ञान या कार्यात्मक संबंध है। (जैसे: पाउंड इंग्लैंड की मुद्रा है)।<br>
    <b>चरण 3:</b> यही कार्यात्मक संबंध "${pair.hi[2]}" पर लागू करें।<br>
    <b>चरण 4:</b> "${pair.hi[2]}" का सही विकल्प "${pair.hi[3]}" है।<br>
    ✅ <b>अतः सही उत्तर "${pair.hi[3]}" है।</b>`;
    
    return { q_en: `${pair.en[0]} : ${pair.en[1]} :: ${pair.en[2]} : ?`, q_hi: `${pair.hi[0]} : ${pair.hi[1]} :: ${pair.hi[2]} : ?`, opts_en, opts_hi, answer: ansIdx, type: "Mixed Analogy", exp_en, exp_hi };
}

// Generate 510
while (questions.length < 510) {
    const type = Math.floor(Math.random() * 4);
    let q;
    if (type === 0) q = generateNumberAnalogy();
    else if (type === 1) q = generateAlphabetAnalogy();
    else if (type === 2) q = generateWordAnalogy();
    else q = generateMixedAnalogy();
    const exists = questions.some(ex => ex.q_en === q.q_en && ex.type === q.type);
    if (!exists) questions.push(q);
}
shuffle(questions);

const container = document.getElementById('questionsContainer');
const search = document.getElementById('search');
const filter = document.getElementById('filter');
const attemptText = document.getElementById('attemptText');
const scoreText = document.getElementById('scoreText');
const progressBar = document.getElementById('progressBar');
let attempted = new Set();
let score = 0;

function renderQuestions() {
    container.innerHTML = '';
    const keyword = search.value.toLowerCase().trim();
    const selectedType = filter.value;
    questions.forEach((q, idx) => {
        if (selectedType !== 'all' && q.type !== selectedType) return;
        const qText = currentLang === 'en' ? q.q_en : q.q_hi;
        const opts = currentLang === 'en' ? q.opts_en : q.opts_hi;
        const exp = currentLang === 'en' ? q.exp_en : q.exp_hi;
        let full = (qText + opts.join(' ') + exp).toLowerCase();
        if (keyword && !full.includes(keyword)) return;

        const card = document.createElement('div');
        card.className = 'question-card';
        card.id = 'question-' + idx;
        let optHTML = '';
        opts.forEach((opt, oi) => {
            optHTML += `<button class="option" onclick="checkAnswer(${idx},${oi},this)">(${String.fromCharCode(65+oi)}) ${opt}</button>`;
        });
        card.innerHTML = `
            <div class="question-top">
                <span class="question-number">Q${idx+1}</span>
                <span class="question-type">${q.type}</span>
            </div>
            <div class="question-body">
                <div class="question-text">${qText}</div>
                <div class="options">${optHTML}</div>
                <div class="answer-result" id="result-${idx}"></div>
                <div class="explanation" id="explanation-${idx}">
                    <div class="explanation-title">${currentLang === 'en' ? '💡 Detailed Explanation (Step-by-Step)' : '💡 विस्तृत व्याख्या (चरण-दर-चरण)'}</div>
                    ${exp}
                </div>
            </div>
        `;
        container.appendChild(card);
    });
}

function checkAnswer(qIdx, selIdx, btn) {
    if (attempted.has(qIdx)) return;
    attempted.add(qIdx);
    const q = questions[qIdx];
    const card = document.getElementById('question-' + qIdx);
    const btns = card.querySelectorAll('.option');
    btns.forEach((b, i) => { if (i === q.answer) b.classList.add('correct'); });
    const res = document.getElementById('result-' + qIdx);
    if (selIdx === q.answer) { score++; res.innerHTML = currentLang === 'en' ? '✅ Correct Answer!' : '✅ सही उत्तर!'; res.style.color = '#16a34a'; }
    else { btn.classList.add('wrong'); res.innerHTML = currentLang === 'en' ? '❌ Incorrect. See explanation below.' : '❌ गलत। नीचे व्याख्या देखें।'; res.style.color = '#dc2626'; }
    document.getElementById('explanation-' + qIdx).classList.add('show');
    updateProgress();
}

function updateProgress() {
    attemptText.textContent = `${currentLang === 'en' ? 'Attempted' : 'हल किए'}: ${attempted.size} / ${questions.length}`;
    scoreText.textContent = `${currentLang === 'en' ? 'Score' : 'स्कोर'}: ${score}`;
    progressBar.style.width = (attempted.size / questions.length * 100) + '%';
}

function showAllExplanations() {
    document.querySelectorAll('.explanation').forEach(e => e.classList.add('show'));
}
function resetQuiz() { attempted.clear(); score = 0; renderQuestions(); updateProgress(); window.scrollTo({top:0, behavior:'smooth'}); }
search.addEventListener('input', renderQuestions);
filter.addEventListener('change', renderQuestions);

updateUI();

/* =====================================================

===================================================== */

function createCategories(){

    const categories =
    [
        ...new Set(
            books.map(
                book =>
                book.category
            )
        )
    ];


    categories.forEach(
        category => {


            /* SUBJECT DROPDOWN */

            const option =
            document.createElement(
                "option"
            );


            option.value =
            category;


            option.textContent =
            category;


            subjectFilter.appendChild(
                option
            );



            /* CATEGORY BUTTON */

            const button =
            document.createElement(
                "button"
            );


            button.className =
            "category";


            button.dataset.category =
            category;


            button.textContent =
            category;


            categoryButtons.appendChild(
                button
            );

        }
    );



    /* CATEGORY CLICK */

    document
    .querySelectorAll(
        ".category"
    )
    .forEach(
        button => {


            button.addEventListener(
                "click",
                function(){


                    document
                    .querySelectorAll(
                        ".category"
                    )
                    .forEach(
                        btn =>
                        btn.classList
                        .remove(
                            "active"
                        )
                    );


                    this.classList
                    .add(
                        "active"
                    );


                    selectedCategory =
                    this.dataset.category;


                    subjectFilter.value =
                    selectedCategory;


                    renderBooks();

                }
            );

        }
    );

}

/* =====================================================
   RENDER BOOKS
===================================================== */

function renderBooks(){


    let result =
    books.filter(
        book => {


            const search =
            searchInput.value
            .toLowerCase()
            .trim();


            const matchesSearch =

            book.title
            .toLowerCase()
            .includes(
                search
            )

            ||

            book.category
            .toLowerCase()
            .includes(
                search
            )

            ||

            book.description
            .toLowerCase()
            .includes(
                search
            );



            const matchesCategory =

            selectedCategory ===
            "all"

            ||

            book.category ===
            selectedCategory;



            const selectedLanguage =
            languageFilter.value;


            const matchesLanguage =

            selectedLanguage ===
            "all"

            ||

            book.language ===
            selectedLanguage;



            return (

                matchesSearch

                &&

                matchesCategory

                &&

                matchesLanguage

            );

        }
    );



    /* =================================================
       SORT A-Z
    ================================================= */

    if(
        sortFilter.value ===
        "az"
    ){

        result.sort(
            (a,b) =>
            a.title.localeCompare(
                b.title
            )
        );

    }



    /* =================================================
       SORT Z-A
    ================================================= */

    if(
        sortFilter.value ===
        "za"
    ){

        result.sort(
            (a,b) =>
            b.title.localeCompare(
                a.title
            )
        );

    }



    /* CLEAR */

    bookGrid.innerHTML =
    "";



    /* NO RESULT */

    if(
        result.length ===
        0
    ){

        noResult.style.display =
        "block";

        return;

    }



    noResult.style.display =
    "none";



    /* DISPLAY */

    result.forEach(
        book => {

            bookGrid.innerHTML +=
            createBookCard(
                book
            );

        }
    );

}
